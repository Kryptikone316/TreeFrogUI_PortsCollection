# Water Closet settings (plain text; edit and relaunch, no rebuild needed).
#
# ---- Speed -------------------------------------------------------------------
#
# fps:        the frame limit, which is also the game speed. This game does one logic
#             step per frame, so 60 is the speed it was designed for (its on-screen
#             clock counts 60 frames to the second); a lower number makes everything
#             slower, not just less smooth. 50 leaves some room on a slower device.
fps=50
#
# draw_every: 1 draws a picture on every frame. 2 or 3 runs the game at the full fps
#             speed but draws only every 2nd or 3rd frame. Use it if the game cannot
#             draw fast enough on your device: it keeps the game speed right, at the
#             price of a choppier picture.
draw_every=1
#
# ---- Effects (turn off to save work on a slow device) -----------------------
#
# lights:     1 shows the soft glows around coins, traffic lights and Walter, 0 hides
#             them. They are the most expensive things drawn.
lights=1
# particles:  1 shows particle effects (splashes and sparkles), 0 hides them.
particles=1
# background: 1 shows the dim brick wall behind the level, 0 leaves it plain dark.
background=1
#
# ---- Picture -----------------------------------------------------------------
#
# The game's pictures are drawn at half size for this screen. These settings tune
# how they are reduced and coloured. Glows and particles are not affected.
#
# sharpness:  how crisp the reduced pictures are. 0 is the softest (a plain 2x2
#             average), 1 the crispest (single pixels picked, like a nearest-pixel
#             scale). Try 0.3 to 0.8.
sharpness=0.5
# saturation: colour strength. 1 is unchanged, above 1 is more colourful.
saturation=1.0
# contrast:   difference between light and dark. 1 is unchanged, above 1 is more.
contrast=1.0
#
# ---- Log ---------------------------------------------------------------------
#
# perf_log:   1 writes a timing line to log.txt every couple of seconds, which helps
#             when reporting a speed problem. 0 turns it off.
perf_log=1

# ---- Priority ----------------------------------------------------------------
#
# nice:       scheduling priority of the game, from -20 (most favoured) to 19. 0
#             treats the game like any other program. A negative value stops small
#             background jobs from taking a slice of a frame, which can show up as
#             an occasional hitch.
nice=-10

# ---- Sound -------------------------------------------------------------------
#
# music:      1 plays the music, 0 turns it off completely (sound effects stay).
#             Streaming the music takes a share of the processor, so turning it
#             off can help on a slow device.
music=1

# audio_lead: how many milliseconds ahead of the clock the sound is mixed, 0 to 400.
#             A bigger lead rides out longer pauses in the game without a gap or
#             crackle in the sound, but every sound effect starts that much later.
audio_lead=60

# ---- Controls ----------------------------------------------------------------
#
# The buttons themselves are set in controls.txt.
#
# jump_repeat: 0 means each jump needs a fresh press of the jump button (a press
#             made while still in the air counts when Walter lands). 1 is how the
#             original game behaves: keep the button held and Walter jumps again
#             the moment he lands.
jump_repeat=0
